javascriptfunction insertTrainingRecord(data) {
  const sheet = getSheet('BotTrainingData');
  
  // Визначити чи це SELF режим
  const isSelf = (data.studentId === "SELF") || 
                 (data.chatId === data.coachId && data.studentId === null);
  
  const row = [
    data.chatId,                    // A: ChatID (Coach в SELF режимі)
    new Date(),                     // B: Date
    data.exerciseName,              // C: ExerciseName
    data.weight,                    // D: Weight
    data.reps,                      // E: Reps
    isSelf ? "SELF" : data.studentId, // F: StudentID
    data.coachId,                   // G: CoachID
    data.trainingId                 // H: TrainingId (з префіксом SELF_)
  ];
  
  sheet.appendRow(row);
  SpreadsheetApp.flush();
}

 * Інвалідувати кілька cache keys одночасно

function invalidateCacheBatch(keys) {
  const cache = CacheService.getScriptCache();
  const removed = [];
  
  keys.forEach(key => {
    try {
      cache.remove(key);
      removed.push(key);
    } catch (error) {
      Logger.log(`[CACHE] Failed to remove ${key}: ${error}`);
    }
  });
  
  Logger.log(`[CACHE] Batch invalidated ${removed.length} keys`);
  return removed;
}

9.11. Вартість тренувань (Pricing) та дохід у звітах

**Встановлення вартості:** Тренер (Мої учні) → Ввести вартість / Індивідуальна вартість / Змінити вартість → FSM (Registration): вибір типу (персональне/спліт/тріо) та введення суми (ціле число, UAH) → Sheets.setPricing() → лист **Pricing** (CoachID, StudentID порожній або конкретний, PricePersonal/PriceSplit/PriceTrio, Currency, UpdatedAt).

**Запис вартості при COMPLETED:** При переході слоту в COMPLETED (Training.finishWorkout_ або Schedule.confirmAttendance): для кожного слоту Sheets.getCurrentPrice(coachId, studentId, coachTrainingType) з листа Pricing → поділ на 1/2/3 за типом (персональне/спліт/тріо) → Sheets.updateScheduleSlotPrice(slotId, perStudentAmount, currency). Заповнюються колонки **WorkoutSchedule.I** (PriceCharged), **WorkoutSchedule.J** (Currency).

**Звіт за доходами:** Звіт «Кількість тренувань» / дохід за період: вибірка COMPLETED слотів за період; сума по полю **PriceCharged** (колонка I) — загальний дохід; при потребі групування по учнях — сума PriceCharged по StudentID.

javascriptfunction getCurrentActiveWorkout(chatId) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID)
    .getSheetByName('BotTrainingData');
  
  const data = sheet.getRange('B3:H').getValues(); // Date to ChatID
  
  const now = new Date();
  const yesterday = new Date(now.getTime() - 24 * 60 * 60 * 1000);
  
  const recentSets = [];
  for (let i = 0; i < data.length; i++) {
    const date = data[i][0];  // B: Date
    const chatIdRow = data[i][6];  // H: ChatID
    
    if (chatIdRow === chatId && date >= yesterday) {
      recentSets.push({
        date: date,
        exerciseId: data[i][1],     // C
        exerciseName: data[i][2],   // D
        weight: data[i][3],         // E
        reps: data[i][4],           // F
        set: data[i][5]             // G
      });
    }
  }
  
  if (recentSets.length === 0) return null;
  
  // Групування по вправах
  return {
    isActive: true,
    startTime: recentSets[0].date,
    exercises: groupByExercise(recentSets)
  };
}

Алгоритм:
javascriptfunction getWorkoutHistory(chatId, limit, filter) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID)
    .getSheetByName('BotTrainingData');
  
  const data = sheet.getRange('A3:H').getValues();
  
  // Фільтрація по ChatID
  let filtered = data.filter(row => row[7] === chatId);  // H: ChatID
  
  // Застосування додаткового фільтру
  if (filter.type === 'by_group') {
    // Потрібно JOIN з ExerciseLibrary по ExerciseID
    const exercises = Sheets.getExercisesByGroup(filter.groupName);
    const exerciseIds = exercises.map(ex => ex.id);
    
    filtered = filtered.filter(row => 
      exerciseIds.includes(row[2])  // C: ExerciseID
    );
  } else if (filter.type === 'by_exercise') {
    filtered = filtered.filter(row => 
      row[2] === filter.exerciseId  // C: ExerciseID
    );
  }
  
  // Групування по датах
  const grouped = groupByDate(filtered);
  
  // Сортування по даті (новіші першими)
  const sorted = grouped.sort((a, b) => b.date - a.date);
  
  // Обмеження кількості
  return sorted.slice(0, limit);
}

3. getWorkoutDetails(chatId, date)
Призначення: Отримати детальну інформацію про одне тренування
Алгоритм:
javascriptfunction getWorkoutDetails(chatId, date) {
  const sheet = SpreadsheetApp.openById(SPREADSHEET_ID)
    .getSheetByName('BotTrainingData');
  
  const data = sheet.getRange('A3:H').getValues();
  
  // Початок та кінець дня
  const startOfDay = new Date(date);
  startOfDay.setHours(0, 0, 0, 0);
  
  const endOfDay = new Date(date);
  endOfDay.setHours(23, 59, 59, 999);
  
  // Фільтрація
  const workout = data.filter(row => {
    const rowDate = row[1];  // B: Date
    const rowChatId = row[7];  // H: ChatID
    
    return rowChatId === chatId && 
           rowDate >= startOfDay && 
           rowDate <= endOfDay;
  });
  
  // Групування по вправах
  const exercises = groupByExercise(workout);
  
  // Групування по групах м'язів
  const byMuscleGroup = groupByMuscleGroup(exercises);
  
  // Статистика
  const stats = {
    totalExercises: exercises.length,
    totalSets: workout.length,
    totalReps: workout.reduce((sum, row) => sum + row[5], 0),  // F: Reps
    duration: calculateDuration(workout)
  };
  
  return {
    date: date,
    muscleGroups: byMuscleGroup,
    stats: stats
  };
}
